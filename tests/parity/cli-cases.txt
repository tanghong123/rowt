# server / sub: the arms that WRITE. What these prove is on disk, not on stdout
# — manual.json, subs.txt, the rebuilt servers.json and the `selected` line of
# state — which is why cli-diff compares the config tree (content and mode) and
# not just the two output streams.
#
# The subscriptions in the `pool-seeded` fixture point at a closed local port,
# so every rebuild here takes the fetch-failed path. That is the only
# subscription path the two implementations can be compared on: bash fetches
# with urllib inside vless-parse.py and rowt-rs with curl, so a SUCCESSFUL fetch
# would differ in the argv trace whatever the bodies were. The parse side of a
# real subscription body is covered by `parity vless-diff`.
server add vless://00000000-0000-4000-8000-000000000009@203.0.113.20:443?security=tls&sni=delta.example#delta
server add
server add notalink
server add vless://00000000-0000-4000-8000-000000000009@203.0.113.20:443#d1 vless://00000000-0000-4000-8000-000000000009@203.0.113.20:443#d2
# Re-importing something the manual set already has must be idempotent on disk,
# not merely deduped again in servers.json on the next rebuild.
PARITY_SCENARIO=pool-seeded server add vless://00000000-0000-4000-8000-000000000001@203.0.113.10:443#alpha-again
server rm alpha
server rm nosuchtag

# A manual set that is not a JSON array at all: `jq -s add` fails and the
# rebuild silently continues without it, so servers.json is rebuilt from the
# subscriptions alone rather than the command failing.

# An array whose elements are not all objects. This one does NOT degrade
# quietly: the Python calls .get on a string, which is an AttributeError rather
# than a ValueError, so it escapes its own handler — the rebuild dies, and the
# audit log gets an ABORT with no END. Both halves of that are compared.
PARITY_SCENARIO=pool-corrupt server list
PARITY_SCENARIO=pool-corrupt server add vless://00000000-0000-4000-8000-000000000009@203.0.113.20:443#delta
PARITY_SCENARIO=pool-corrupt server clear
